    if (best < 0) return;
    ev.preventDefault(); ev.stopPropagation();
    var i = best;
    var startX = ev.clientX;
    var startPct = pcts[i], pairPct = pcts[i] + pcts[i + 1];
    var tw = t.offsetWidth || 1;
    var minPct = 45 / tw * 100;
    function move(e){
      var w = startPct + (e.clientX - startX) / tw * 100;
      w = Math.max(minPct, Math.min(pairPct - minPct, w));
      pcts[i] = w; pcts[i + 1] = pairPct - w;
      apply(); lastDrag = Date.now();
    }
    function up(){
      document.removeEventListener('pointermove', move);
      document.removeEventListener('pointerup', up);
      save();
    }
    document.addEventListener('pointermove', move);
    document.addEventListener('pointerup', up);
  });
  ths.forEach(function(th, i){
    th.addEventListener('click', function(){
      if (Date.now() - lastDrag < 250) return;
      var asc = !th.classList.contains('sasc');
      ths.forEach(function(h){ h.classList.remove('sasc', 'sdesc'); });
      th.classList.add(asc ? 'sasc' : 'sdesc');
      var rows = Array.prototype.slice.call(t.querySelectorAll('tbody tr'));
      rows.sort(function(a, b){
        var x = a.children[i], y = b.children[i];
        var xv = x.hasAttribute('data-v') ? x.getAttribute('data-v') : x.textContent.trim();
        var yv = y.hasAttribute('data-v') ? y.getAttribute('data-v') : y.textContent.trim();
        var xn = parseFloat(xv), yn = parseFloat(yv);
        var c = (!isNaN(xn) && !isNaN(yn)) ? xn - yn : String(xv).localeCompare(String(yv));
        return asc ? c : -c;
      });
      var tb = t.querySelector('tbody');
      rows.forEach(function(r){ tb.appendChild(r); });
    });
  });
});
