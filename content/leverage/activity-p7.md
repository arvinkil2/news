document.querySelectorAll('table.bw-grid').forEach(function(t){

  var n = ths.length;

      var d = Math.abs(ev.clientX - ths[i].getBoundingClientRect().right);

      else widths[i] = Math.max(mins[i], startW + dx);

  });
