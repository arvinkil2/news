      var blob = new Blob([lines.join('\n')], {type: 'text/csv'});
      var a = document.createElement('a');
      a.href = URL.createObjectURL(blob);
      a.download = btn.getAttribute('data-name') || 'table.csv';
      a.click();
      URL.revokeObjectURL(a.href);
    });
  });
})();
